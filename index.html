Button Hover Animation

01 · Slide Fill
.btn::before {
  content: "";
  position: absolute;
  top: 0;
  left: 0;
  width: 0%;
  height: 100%;
  background: #22c55e;
  transition: 0.4s;
  z-index: -1;
}
02 · Border Draw
.btn::before,
.btn::after {
  content: "";
  position: absolute;
  width: 0;
  height: 2px;
  background: #f59e0b;
  transition: .3s;
}

/* Hover expands the top and bottom borders */
.btn:hover::before,
.btn:hover::after {
  width: 100%;
}
03 · Shine Sweep
.btn:hover::after {
  animation: shine .8s;
}

@keyframes shine {
  100% { left: 150%; }
}
04 · Gradient Move
.btn:hover {
  animation: gradientMove 2s linear infinite;
}

@keyframes gradientMove {
  0% { background-position: 0%; }
  100% { background-position: 300%; }
}
05 · Ripple Hover
.btn:hover {
  box-shadow: 0 0 0 10px
    rgba(16,185,129,0.3);
}
06 · Neon Border
.btn:hover {
  box-shadow:
    0 0 10px #0ff,
    0 0 20px #0ff,
    0 0 40px #0ff;
}
07 · Flip Button
.btn:hover {
  transform: rotateX(360deg);
}
08 · Glow Button
